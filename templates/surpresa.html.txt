        }
    });

// CRIAR POLAROIDS
for(let i=1;i<=12;i++){
    let p = document.createElement("div");
    p.className = "polaroid";

    p.style.top = (300 + i*250) + "px";
    p.style.left = (Math.random()*60 + 20) + "%";
    p.style.transform = "rotate(" + (Math.random()*20 - 10) + "deg)";

    p.innerHTML = `<img src="/static/fotos/foto${i}.jpg">`;

    document.body.appendChild(p);
}

// BALÕES COM PADRÃO
const padrao = [5,3,1,4,2];
let index = 0;

function criarBaloes(){
    let qtd = padrao[index];
    index = (index + 1) % padrao.length;

    for(let i=0;i<qtd;i++){
        let b = document.createElement("div");
        b.className = "balloon";
        b.innerHTML = "💗";

        b.style.left = Math.random()*100 + "vw";

        b.onclick = () => {
            for(let i=0;i<15;i++){
                let mini = document.createElement("div");
                mini.innerHTML = "❤️";
                mini.style.position = "fixed";
                mini.style.left = b.style.left;
                mini.style.top = "50%";
                mini.style.transform = `translate(${Math.random()*200-100}px, ${Math.random()*200-100}px)`;
                document.body.appendChild(mini);
                setTimeout(()=>mini.remove(),1000);
            }
            b.remove();
        };

        document.body.appendChild(b);
        setTimeout(()=>b.remove(),8000);
    }
}

setInterval(criarBaloes, 3000);
